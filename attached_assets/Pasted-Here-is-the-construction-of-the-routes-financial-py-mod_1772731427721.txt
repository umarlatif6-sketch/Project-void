import numpy as np

def manual_hilbert(x):
    """Replacement for scipy.signal.hilbert using the analytic signal method."""
    N = len(x)
    Xf = np.fft.fft(x)
    h = np.zeros(N)
    if N % 2 == 0:
        h[0] = h[N//2] = 1
        h[1:N//2] = 2
    else:
        h[0] = 1
        h[1:(N+1)//2] = 2
    return np.fft.ifft(Xf * h)
