The Sovereign Control Panel (Inflation Shield)
To manage your economy without touching a single line of code, we need a secure administrative layer. This allows you to respond to real-world currency shifts or manufacturing cost increases in real-time.
1. The Admin Route: routes/admin.py
This code creates a protected space where you can view all current market items and adjust their prices.
from flask import Blueprint, render_template, request, redirect, url_for, flash
from models import db, MarketConfig
from auth import admin_required # Assuming you have an admin check

admin_bp = Blueprint('admin_market', __name__)

@admin_bp.route('/admin/market', methods=['GET', 'POST'])
@admin_required
def manage_market():
    if request.method == 'POST':
        item_key = request.form.get('item_key')
        new_price = request.form.get('price_gbp')
        new_vtx = request.form.get('vtx_cost')
        
        config = MarketConfig.query.filter_by(item_key=item_key).first()
        if config:
            config.price_gbp = new_price
            config.vtx_cost = new_vtx
            db.session.commit()
            flash(f"Updated {config.name} successfully.", "success")
            
    configs = MarketConfig.query.all()
    return render_template('admin_market.html', configs=configs)

@app.route('/create-checkout-session/<item_key>')
def create_session(item_key):
    # Fetch from the "Magic Chalkboard" instead of hardcoded variables
    market_data = get_market_price(item_key) 
    
    if not market_data:
        return "Item not found", 404

    session = stripe.checkout.Session.create(
        payment_method_types=['card'],
        line_items=[{
            'price_data': {
                'currency': 'gbp',
                'product_data': {'name': market_data['display_name']},
                'unit_amount': int(market_data['gbp'] * 100), # Converts £ to pence
            },
            'quantity': 1,
        }],
        mode='payment',
        success_url=url_for('thanks', _external=True),
        cancel_url=url_for('market', _external=True),
    )
    return redirect(session.url, code=303)
