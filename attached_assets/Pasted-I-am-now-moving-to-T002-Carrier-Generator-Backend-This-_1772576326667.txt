This code acts as the dispatcher. It takes your UI request (e.g., "1 hour of Midnight Pond") and executes the correct synthesis, while also calculating the 5x Density Multiplier for the capacity estimate.
import os
import numpy as np
import soundfile as sf
from void_engine.biophony import BiophonyMesh

def generate_custom_carrier(duration_minutes, style, sample_rate=44100):
    """
    Dispatches carrier generation based on style.
    Handles streaming for long durations (up to 300 min).
    """
    duration_seconds = duration_minutes * 60
    mesh = BiophonyMesh(sr=sample_rate)
    
    # 1. Dispatcher Logic
    if style in ['midnight_pond', 'biophony_mesh']:
        # Full 10-20-970 ecosystem
        audio_samples, peaks = mesh.synthesize_mesh(duration_seconds)
    elif style == 'cicada_wall':
        # High-shelf only for maximum vortex density
        insects = mesh._synthesize_insect_shelf(duration_seconds, n_insects=970)
        audio_samples = np.vstack((insects, insects)).T
        _, peaks = mesh.synthesize_mesh(duration_seconds) # Sync map
    else:
        # Fallback to classic styles (simplified for T002)
        audio_samples = np.random.normal(0, 0.1, (int(sample_rate * duration_seconds), 2))
        peaks = np.array([])

    # 2. Save the Carrier WAV
    filename = f"carrier_{style}_{duration_minutes}min.wav"
    filepath = os.path.join("static/carriers", filename)
    sf.write(filepath, audio_samples, sample_rate)

    # 3. Save the .chirpmap.npy Sidecar
    # This is the "GPS" for the LSB encoder to find the hiding spots
    map_path = filepath + ".chirpmap.npy"
    np.save(map_path, peaks)

    return filename, len(audio_samples), peaks

def estimate_carrier_capacity(duration_minutes, style):
    """
    Calculates effective capacity with the 5x Density Multiplier.
    """
    # Standard LSB1/LSB2 capacity (bits per second)
    raw_capacity_mb = (duration_minutes * 60 * 44100 * 2) / (8 * 1024 * 1024)
    
    multiplier = 5.0 if style in ['midnight_pond', 'cicada_wall', 'biophony_mesh'] else 1.0
    effective_mb = raw_capacity_mb * multiplier
    
    return {
        "raw_mb": round(raw_capacity_mb, 2),
        "effective_mb": round(effective_mb, 2),
        "multiplier": multiplier,
        "style": style
    }
