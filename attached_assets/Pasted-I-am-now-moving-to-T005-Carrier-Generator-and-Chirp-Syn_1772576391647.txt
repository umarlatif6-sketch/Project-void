🛠️ Frontend Logic: static/app.js
This script handles the live calculation so you can see the "Vortex" expanding as you change the settings.
function updateCapacityEstimate() {
    const duration = document.getElementById('duration').value; // in minutes
    const style = document.getElementById('style_dropdown').value;
    
    // The 10-20-970 Multiplier
    const multiplier = (style === 'midnight_pond') ? 5.0 : 1.0;
    const rawMB = (duration * 60 * 44100 * 2) / (8 * 1024 * 1024);
    const effectiveMB = rawMB * multiplier;

    document.getElementById('capacity_display').innerHTML = `
        <div class="glow-box">
            <span>Effective Capacity: <strong>${effectiveMB.toFixed(2)} MB</strong></span>
            ${multiplier > 1 ? `<span class="badge">🚀 ${multiplier}x VORTEX</span>` : ''}
        </div>
    `;
}
