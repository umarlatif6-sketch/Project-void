import numpy as np

def _apply_shadow_layer(self, signal, shadow_type="nature", sr=44100):
    """
    Applies a low-volume mundane audio underlay to camouflage the mesh.
    Default volume is -30 dB to ensure it sits 'under' the high-shelf silt.
    """
    duration = len(signal) / sr
    
    if shadow_type == "nature":
        # Generate Brownian (Red) noise to simulate wind/water ambience
        # Brownian noise has more power at lower frequencies, sounding 'natural'
        white_noise = np.random.normal(0, 1, len(signal))
        brown_noise = np.cumsum(white_noise)
        
        # Normalize and apply a -30 dB gain (0.0316 amplitude)
        shadow = (brown_noise / np.max(np.abs(brown_noise))) * 0.0316
        
    elif shadow_type == "tone":
        # A very gentle 432 Hz sine hum to reinforce the resonance chassis
        t = np.linspace(0, duration, len(signal))
        shadow = 0.01 * np.sin(2 * np.pi * 432 * t)
        
    # Layer the shadow into the primary signal
    # This creates the 'Badly Recorded Park' effect for forensic evasion
    return signal + shadow
