import sqlite3
import json
from datetime import datetime

class RootChronicle:
    def __init__(self, db_path="void_lineage.db"):
        self.conn = sqlite3.connect(db_path, check_same_thread=False)
        self._initialize_db()

    def _initialize_db(self):
        cursor = self.conn.cursor()
        # The 'Lineage' table stores successful Al-Jabr patterns
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS lineage (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT,
                sensor_snapshot TEXT,
                winning_roots TEXT,
                slm_rating REAL,
                is_founder_wisdom INTEGER DEFAULT 0
            )
        ''')
        self.conn.commit()

    def record_wisdom(self, sensors, roots, slm, is_founder=0):
        """Saves a 'Morphic Truth' to the Chronicle."""
        cursor = self.conn.cursor()
        cursor.execute('''
            INSERT INTO lineage (timestamp, sensor_snapshot, winning_roots, slm_rating, is_founder_wisdom)
            VALUES (?, ?, ?, ?, ?)
        ''', (datetime.now().isoformat(), json.dumps(sensors), roots, slm, is_founder))
        self.conn.commit()

    def recall_wisdom(self, current_sensors, threshold=0.1):
        """Queries 'Ancestral Memory' for similar past situations."""
        cursor = self.conn.cursor()
        cursor.execute('SELECT sensor_snapshot, winning_roots FROM lineage WHERE slm_rating > 0.95')
        past_events = cursor.fetchall()
        
        # Simple Pattern Match: If current temp/power matches a past 'Winner', return it.
        for snapshot_json, roots in past_events:
            past_sensors = json.loads(snapshot_json)
            # Logic: If Temp and Power are within 10% of past successful states
            if abs(current_sensors['temp'] - past_sensors['temp']) < (past_sensors['temp'] * threshold):
                return roots  # This is the 'Proven Root'
        return None
