import time
import uuid

class MeshPacket:
    def __init__(self, source_id, dest_id, payload, hops=0):
        self.packet_id = str(uuid.uuid4())[:8]
        self.source_id = source_id
        self.dest_id = dest_id  # Use 'BROADCAST' for discovery
        self.payload = payload
        self.hops = hops
        self.max_hops = 7  # The 'Seven Seas' Limit
        self.timestamp = time.time()

class MeshRouter:
    def __init__(self, node_id):
        self.node_id = node_id
        self.routing_table = {} # {neighbor_id: {signal_strength, last_seen, hops}}
        self.seen_packets = set()

    def process_packet(self, packet):
        """Logic for relaying or consuming a packet."""
        # 1. Prevent infinite loops
        if packet.packet_id in self.seen_packets:
            return None, "Already seen"
        
        self.seen_packets.add(packet.packet_id)

        # 2. Check if the packet is for US
        if packet.dest_id == self.node_id:
            return packet.payload, "Destination reached"

        # 3. Relay Logic (The 'Seven Seas' Rule)
        if packet.hops < packet.max_hops:
            packet.hops += 1
            # Relaying costs 0.2 CC (Managed by T003 Integration)
            return packet, f"Relaying to hop {packet.hops}"
        
        return None, "Max hops exceeded"

    def update_routing(self, neighbor_id, strength):
        """Updates the local map of who is nearby."""
        self.routing_table[neighbor_id] = {
            "strength": strength,
            "last_seen": time.time()
        }
