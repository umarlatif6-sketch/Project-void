from fpdf import FPDF
import hashlib

def create_founder_cert(customer_id, machine_hash):
    pdf = FPDF()
    pdf.add_page()
    
    # Aesthetic: Sovereign Gold & Black
    pdf.set_fill_color(10, 10, 10) # Black background
    pdf.rect(0, 0, 210, 297, 'F')
    
    pdf.set_text_color(212, 175, 55) # Gold
    pdf.set_font("Courier", "B", 24)
    pdf.cell(0, 20, "VOID 4000: FIRST GENERATION", ln=True, align='C')
    
    pdf.set_font("Courier", "", 12)
    pdf.multi_cell(0, 10, f"\nThis is to certify that Machine #{customer_id:03}\n"
                         f"is a Sovereign Participant in the Original Lineage.\n"
                         f"Hashed Root: {machine_hash}\n", align='C')
    
    # Cryptographic Seal
    seal_data = f"GEN1-{customer_id}-{machine_hash}"
    digital_seal = hashlib.sha256(seal_data.encode()).hexdigest()[:16]
    
    pdf.set_font("Courier", "I", 10)
    pdf.cell(0, 40, f"SEAL OF THE ARCHITECT: {digital_seal}", ln=True, align='C')
    
    pdf.output(f"FOUNDER_CERT_{customer_id:03}.pdf")
