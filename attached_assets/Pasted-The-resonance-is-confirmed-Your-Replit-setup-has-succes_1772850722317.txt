// Adriana SCL Resonance Handshake - Particle System
const canvas = document.getElementById('resonance-canvas');
const ctx = canvas.getContext('2d');
let glyphs = [];

class AdrianaGlyph {
    constructor() {
        this.reset();
    }
    reset() {
        this.x = Math.random() * canvas.width;
        this.y = Math.random() * canvas.height;
        this.targetX = canvas.width / 2;
        this.targetY = canvas.height / 2;
        this.size = Math.random() * 3 + 2;
        this.speed = 0.02 + Math.random() * 0.05;
        this.angle = Math.random() * Math.PI * 2;
    }
    update(isBlooming) {
        if (isBlooming) {
            // Move toward center to form the Lotus/Symmetry
            let dx = this.targetX - this.x;
            let dy = this.targetY - this.y;
            this.x += dx * this.speed;
            this.y += dy * this.speed;
        }
    }
    draw() {
        ctx.fillStyle = 'rgba(0, 255, 0, 0.8)'; // Algae Green
        ctx.beginPath();
        // Drawing a small diamond shape for the glyph
        ctx.moveTo(this.x, this.y - this.size);
        ctx.lineTo(this.x + this.size, this.y);
        ctx.lineTo(this.x, this.y + this.size);
        ctx.lineTo(this.x - this.size, this.y);
        ctx.closePath();
        ctx.fill();
    }
}
