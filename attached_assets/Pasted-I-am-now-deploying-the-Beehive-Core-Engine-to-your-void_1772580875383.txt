import numpy as np
from scipy.fft import fft

class BeehiveProtocol:
    def __init__(self, resonance_freq=432, sr=44100):
        self.resonance_freq = resonance_freq
        self.sr = sr
        self.nodes = {} # {node_id: metadata}
        self.mesh_state = "SCANNING"

    def generate_handshake_pulse(self, passphrase, duration=1.0):
        """
        Generates a 432Hz 'Sapphire Thread' pulse.
        The secret is in the Phase: derived from the passphrase.
        """
        t = np.linspace(0, duration, int(self.sr * duration))
        
        # Determine secret phase (0 to 2*pi) from passphrase hash
        secret_phase = (int(hash(passphrase)) % 360) * (np.pi / 180)
        
        # Generate the 'Shout' pulse
        pulse = np.sin(2 * np.pi * self.resonance_freq * t + secret_phase)
        
        # Wrap in 'Insect Silt' camouflage
        silt = np.random.normal(0, 0.05, len(t))
        return (pulse + silt).astype(np.float32)

    def detect_and_authenticate(self, audio_buffer, passphrase):
        """
        FFT-based analysis to find the 432Hz spike and verify its phase.
        """
        n = len(audio_buffer)
        yf = fft(audio_buffer)
        xf = np.fft.fftfreq(n, 1/self.sr)
        
        # 1. Find the 432Hz index
        idx = np.argmin(np.abs(xf - self.resonance_freq))
        magnitude = np.abs(yf[idx])
        
        # 2. SNR Threshold Check (5x mean energy)
        if magnitude < np.mean(np.abs(yf)) * 5:
            return False, "Signal too weak"

        # 3. Phase Authentication
        # Extract the phase angle from the complex FFT result
        detected_phase = np.angle(yf[idx])
        expected_phase = (int(hash(passphrase)) % 360) * (np.pi / 180)
        
        # Normalize phases for comparison
        diff = np.abs(detected_phase - expected_phase) % (2 * np.pi)
        if diff > np.pi:
            diff = 2 * np.pi - diff

        # Tolerance of ±15 degrees (0.26 radians) for drift
        if diff <= 0.26:
            self.mesh_state = "CONNECTED"
            return True, "Authenticated"
        
        return False, "Phase mismatch"
